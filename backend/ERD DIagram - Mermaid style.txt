    EventInfo {
        INTEGER EventID PK
        TEXT eventName
        TEXT eventDate
        TEXT eventType
        REAL eventFee
        REAL healthDeptFee
        REAL mileageReimbursement
        REAL eventRunnerFees
        REAL giftCardSales
        REAL cash
        REAL card
        REAL venmo
        REAL other
        REAL cashApp
        REAL taxOverride
    }

    SalesSummary {
        INTEGER SalesSummaryID PK
        INTEGER EventID FK
        REAL grossSales
        REAL refunds
        REAL discounts
        REAL tips
        REAL totalSales
        REAL squareReportedTax
        REAL squareFees
    }

    DrinkSales {
        INTEGER DrinkSaleID PK
        INTEGER EventID FK
        TEXT drinkName
        INTEGER quantitySold
        REAL unitPrice
        REAL totalCost
    }

    Discounts {
        INTEGER DiscountID PK
        INTEGER EventID FK
        TEXT discountName
        REAL discountAmount
    }

    TipTracker {
        INTEGER TipID PK
        INTEGER EventID FK
        REAL tipAmount
    }

    SupplyCosts {
        INTEGER SupplyID PK
        INTEGER EventID FK
        TEXT itemName
        REAL quantityUsed
        REAL totalCost
    }

    AdditionalFees {
        INTEGER FeeID PK
        INTEGER EventID FK
        TEXT feeName
        REAL feeAmount
    }

    EmployeeTracker {
        INTEGER EmployeeID PK
        TEXT EmployeeName
        REAL defaultWage
    }

    EventEmployees {
        INTEGER EventEmployeeID PK
        INTEGER EventID FK
        INTEGER EmployeeID FK
        REAL hoursWorked
        REAL hourlyRate
        REAL totalPay
    }

    EventPayments {
        TEXT PaymentID PK
        INTEGER EventID FK
        REAL amount
        REAL taxAmount
        REAL feeAmount
        TEXT method
    }
